get_pums_data_as_tibble_numeric_vars_only_by_year <- function(api_key,
                                             year_str = c("2024", "2023", "2022", "2021"), #by putting 2024 as the beginning it becomes a default value when it is not specified 
                                             numeric_vars = c("AGEP", "PWGTP")) {
  #check for missing inputs
  if (missing(api_key)) {
    stop("apiKey are required")
  }
  
  year_str <- match.arg(year_str, several.ok = FALSE)
  
  if (missing(year_str)) {
    message("year_str is not supplied, 2024 is supplied as default")
  }
  variables_supported = load_variables_rds_by_year(year_str)
  
  allowed_numeric_vars <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP")
  non_allowed_nuermic_var_in_input <- FALSE
  is_pwgtp_part_of_input <- FALSE
  number_of_numeric_values_outside_pwgtp <- 0
  
  # validate the input tags
  for (v in numeric_vars) {
    if (v == "PWGTP") {
      is_pwgtp_part_of_input <- TRUE
    }
    v_allowed <- FALSE
    for (allowed in allowed_numeric_vars) {
      if (v == allowed) {
        v_allowed <- TRUE
      }
    }
    
    if (!v_allowed) {
      non_allowed_nuermic_var_in_input <- TRUE
    }
    else if (v != "PWGTP") {
      number_of_numeric_values_outside_pwgtp <- number_of_numeric_values_outside_pwgtp + 1
    }
  }
  
  if (non_allowed_nuermic_var_in_input) {
    stop("numeric_vars must only contain: ",
         paste(allowed_numeric_vars, collapse = ", "))
  }
  
  if (number_of_numeric_values_outside_pwgtp == 0) {
    stop("at least one numeric var other than PWGTP must be specified in the input")
  }
  
  if (!is_pwgtp_part_of_input) {
    numeric_vars <- c(numeric_vars, "PWGTP")
  }
  
  url <- paste0(
    "https://api.census.gov/data/",
    year_str,
    "/acs/acs1/pums?get=",
    paste(numeric_vars, collapse = ","),
    "&SCHL=24&key=",
    api_key
  )
  
  pums_data <- GET(url)
  resp_json <- fromJSON(rawToChar(pums_data$content))
  #resp_tibble <- as_tibble(resp_json)
  
  pums_tibble <- as_tibble(resp_json[-1, , drop = FALSE]) # without -1 header becomes the first row 
  colnames(pums_tibble) <- resp_json[1, ]
  pums_tibble
}

Use lubridate to create a helper method to parse and format time variables

library(lubridate)

parse_time_label_midpoint <- function(label) {
  if(is.null(label) || is.na(label)) {
    return(as.POSIXct(NA))
  }
  
  parts <- strsplit(label, " to ")[[1]]
  
  if(length(parts) != 2) {
    return(as.POSIXct(NA))
  }
  
  start_clean <- gsub("\\.","",parts[1]) #double esacping it for R syntax 
  end_clean   <- gsub("\\.","",parts[2])
  
  start_time <- lubridate::parse_date_time(start_clean, orders = "IMp")
  end_time <- lubridate::parse_date_time(end_clean, orders = "IMp")
  
  if(is.na(start_time) || is.na(end_time)) {
    return(as.POSIXct(NA))
  }
  
  start_time + (end_time - start_time) / 2
}


convert_time_codes_to_midpoints <- function(codes, variable_metadata) {
  result <- sapply(codes, function(code) {
    if(is.na(code) || code == "") {
      return(as.POSIXct(NA))
    }
    
    label <- variable_metadata$values$item[[code]]
    if(is.null(label)) {
      return(as.POSIXct(NA))
    }
    parse_time_label_midpoint(label)
  })
   as.POSIXct(result, origin = "1970-01-01")
}

Let’s write a helper method that helps to match a single value against a set of valid values

match_allowed_value <- function(value, allowed_values) {
  for (allowed in allowed_values) {
    if (toupper(value) == toupper(allowed)) {
      return(allowed)
    }
  }
  return(NA)
}

So far we have only implemented numerical variable queries Let’s start implementing queries that include categorical variables now Also we are adding filters based on geography

get_pums_data_as_tibble_final<-  function(api_key,
                                                                                   year_str = c("2024", "2023", "2022", "2021"),
                                                                                   numeric_vars = c("AGEP", "PWGTP"),
                                                                                   categorical_vars = c("SEX"),
                                                                                   geography_level = c("state"),
                                                                                   geography_value = c("37")) {
  #check for missing inputs
  if (missing(api_key)) {
    stop("apiKey are required")
  }
  
  year_str <- match.arg(year_str, several.ok = FALSE)
  
  if (missing(year_str)) {
    message("year_str is not supplied, 2024 is supplied as default")
  }
  variables_supported <- load_variables_rds_by_year(year_str)
  
  allowed_numeric_vars <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP")
  non_allowed_nuermic_var_in_input <- FALSE
  is_pwgtp_part_of_input <- FALSE
  number_of_numeric_values_outside_pwgtp <- 0
  
  # validate the input tags
  for (v in numeric_vars) {
    if (v == "PWGTP") {
      is_pwgtp_part_of_input <- TRUE
    }
    v_allowed <- FALSE
    for (allowed in allowed_numeric_vars) {
      if (v == allowed) {
        v_allowed <- TRUE
      }
    }
    
    if (!v_allowed) {
      print(v)
      non_allowed_nuermic_var_in_input <- TRUE
    }
    else if (v != "PWGTP") {
      number_of_numeric_values_outside_pwgtp <- number_of_numeric_values_outside_pwgtp + 1
    }
  }
  
  if (non_allowed_nuermic_var_in_input) {
    stop("numeric_vars must only contain: ",
         paste(allowed_numeric_vars, collapse = ", "))
  }
  
  if (number_of_numeric_values_outside_pwgtp == 0) {
    stop("at least one numeric var other than PWGTP must be specified in the input")
  }
  
  if (!is_pwgtp_part_of_input) {
    numeric_vars <- c(numeric_vars, "PWGTP")
  }
  
  
  allowed_categorical_vars <- c("FER", "HHL", "SCH", "SCHL", "SEX") # notice SCHL is a cat variable that can take different values
  non_allowed_categorical_var_in_input <- FALSE
  numb_categorial_variables <- 0
  # validate the input tags
  for (v in categorical_vars) {
    v_allowed <- FALSE
    for (allowed in allowed_categorical_vars) {
      if (v == allowed) {
        v_allowed <- TRUE
        numb_categorial_variables <- numb_categorial_variables + 1
      }
    }
    
    if (!v_allowed) {
      non_allowed_categorical_var_in_input <- TRUE
    }
  }
  
  if (non_allowed_categorical_var_in_input) {
    stop(
      "categorical_vars must only contain: ",
      paste(allowed_categorical_vars, collapse = ", ")
    )
  }
  
  if (numb_categorial_variables == 0) {
    stop(
      "at least one valid categorical must be specified in the input. categorical_vars must only contain: ",
      paste(allowed_categorical_vars, collapse = ", ")
    )
  }

  #let's validate geography
  
  variables_metadata <- get_variables_by_year(year_str)
  
  geography_level <- match_allowed_value(geography_level, c("Region", "Division", "State"))
  if (is.na(geography_level)) {
    stop("geography_level must be one of: Region, Division, State")
  }
  
  if (geography_level == "State") {
    if (year_str == "2021" || year_str == "2022") {
      geography_var_name <- "ST"
    } else {
      geography_var_name <- "STATE"
    }
    geography_for_key <- "state"
  } else if (geography_level == "Division") {
    geography_var_name <- "DIVISION"
    geography_for_key <- "division"
  } else if (geography_level == "Region") {
    geography_var_name <- "REGION"
    geography_for_key <- "region"
  }
  
  
  allowed_geography_values <- names(variables_supported[[geography_var_name]]$values$item)
  
  geography_value <- match_allowed_value(geography_value, c(allowed_geography_values, "all"))
  if (is.na(geography_value)) {
    stop(
      "geography_value must be 'all' or one of the following valid values for ",
      geography_level,
      ": ",
      paste(allowed_geography_values, collapse = ", ")
    )
  }
  
  
  geography_clause <- if (geography_value == "all") {
    paste0("&for=", geography_for_key, ":*")
  } else {
    paste0("&for=", geography_for_key, ":", geography_value)
  }
  
  url <- paste0(
    "https://api.census.gov/data/",
    year_str,
    "/acs/acs1/pums?get=",
    paste(numeric_vars, collapse = ","),
    ",",
    paste(categorical_vars, collapse = ","),
    ",",
    geography_var_name,
    geography_clause,
    "&key=",# &SCHL=24 is now removed
    api_key
  )
  pums_data <- GET(url)
  resp_json <- fromJSON(rawToChar(pums_data$content))
  
  pums_tibble <- as_tibble(resp_json[-1, , drop = FALSE]) # without -1 header becomes the first row
  colnames(pums_tibble) <- resp_json[1, ]
  
  for (col_name in colnames(pums_tibble)) {
    if (col_name %in% c("JWAP", "JWDP")) {
      pums_tibble[[col_name]] <- convert_time_codes_to_midpoints(pums_tibble[[col_name]], variables_supported[[col_name]])
    } else if (col_name %in% allowed_numeric_vars) {
      pums_tibble[[col_name]] <- as.numeric(pums_tibble[[col_name]])
    } else if (col_name %in% allowed_categorical_vars) {
      # adding factors for all categorical values based on distinct values that they support
      pums_tibble[[col_name]] <- factor(pums_tibble[[col_name]],
                                        levels = names(variables_supported[[col_name]]$values$item))
      levels(pums_tibble[[col_name]]) <- unname(unlist(variables_supported[[col_name]]$values$item)) # labels
      
    }
  }
  
  class(pums_tibble) <- c("census", class(pums_tibble))
  pums_tibble
}

Combining tibbles from different years

Now let’s write a function that can take multiple valid years and return combined tibble

get_pums_data_as_tibble_multiple_years <- function(api_key,
                                                   year_strs,
                                                   numeric_vars = c("AGEP", "PWGTP"),
                                                   categorical_vars = c("SEX"),
                                                   geography_level = c("state"),
                                                   geography_value = c("37")) {
  allowed_years <- c("2024", "2023", "2022", "2021")
  for (yr in year_strs) {
    yr_allowed <- FALSE
    for (allowed in allowed_years) {
      if (yr == allowed) {
        yr_allowed <- TRUE
      }
    }
    if (!yr_allowed) {
        stop("year_strs must only contain: ",
             paste(allowed_years, collapse = ", "))
    }
  }
  
  combined_tibble <- NULL
  for (yr in year_strs) {
    year_tibble <- get_pums_data_as_tibble_final(
      api_key,
      yr,
      numeric_vars = numeric_vars,
      categorical_vars = categorical_vars,
      geography_level = geography_level,
      geography_value = geography_value
    )
    
    #normalize ST to STATE to that combining works
    if ("ST" %in% colnames(year_tibble)) {
      colnames(year_tibble)[colnames(year_tibble) == "ST"] <- "STATE"
    }
    
    year_tibble$year <- yr
    
    if (is.null(combined_tibble)) {
      combined_tibble <- year_tibble
    } else {
      combined_tibble <- dplyr::bind_rows(combined_tibble, year_tibble)
    }
  }
  
  class(combined_tibble) <- c("census", class(combined_tibble))
  combined_tibble
  
}

summary.census <- function(x, numeric_vars = NULL, categorical_vars = NULL) {
  
  dflt_categorical_vars <- c("FER", "HHL", "SCH", "SCHL", "SEX","year")
  dflt_numeric_vars <- c("AGEP", "GASP", "GRPIP",  "JWMNP")

  if(is.null(numeric_vars)) {
    numeric_vars = dflt_numeric_vars
  } else {
    for(n in numeric_vars) {
      matched_value <- match_allowed_value(n,dflt_numeric_vars)
      if(is.na(matched_value)) {
        stop("numeric variables must only contain: ", paste(dflt_numeric_vars,collapse = ","))
      }
    }
  }
  
  if(is.null(categorical_vars)) {
    categorical_vars = dflt_categorical_vars
  } else {
    for(c in categorical_vars) {
      matched_value <- match_allowed_value(c,dflt_categorical_vars)
      if(is.na(matched_value)) {
        stop("numeric variables must only contain: ", paste(dflt_categorical_vars,collapse = ","))
      }
    }
  }
  
categorical_vars <- categorical_vars[categorical_vars %in% colnames(x)] 
 numeric_vars <- numeric_vars[numeric_vars %in% colnames(x)]
 
 if(length(categorical_vars) == 0) {
   warning("None of the requested categorical_vars are present in the tibble - no categorical summary will be produced")
 }
 
  if(length(numeric_vars) == 0) {
   warning("None of the requested numeric_vars are present in the tibble - no numerical summary will be produced")
 }
  
  summary_result <- list()
  
  for(v in numeric_vars) {
    
    numeric_vector <- x[[v]]
    weight_vector <- x[["PWGTP"]]
    sample_mean <- sum(numeric_vector * weight_vector) / sum(weight_vector)
    sample_sd = sqrt(sum(numeric_vector^2 * weight_vector)/sum(weight_vector) - sample_mean^2)
    
    summary_result[[v]] <- list(mean = sample_mean, sd = sample_sd)
  }
  
  for(c in categorical_vars) {
    summary_result[[c]] <- table(x[[c]])
  }
  
  summary_result

}

  • It should take one numerical and one caterorical variable as inputs

Before that let’s construct a label query so that I can display it on the boxplot

get_label_for_var <- function(var) {
  
  variables_metadata <- load_variables_rds_by_year("2024") #let's use default 2024
  #I also checked ST and STATE variables between different years the labels are same 
  var_label <- variables_metadata[[var]]$label
  if (is.null(var_label)) {
    var_label <- var   # "year" has no metadata entry -- fall back to the raw name
  }

  var_label
}

Implementing the actual plot function

plot.census <- function(x, numeric_var, categorical_var) {
  if(missing(numeric_var) || missing(categorical_var)) {
    stop("both numeric_var and categorical_var are required")
  }
  
  allowed_categorical_vars <- c("FER", "HHL", "SCH", "SCHL", "SEX","year")
  allowed_numeric_vars <- c("AGEP", "GASP", "GRPIP",  "JWMNP")
  
  numeric_var <- match_allowed_value(numeric_var, allowed_numeric_vars)
  if(is.na(numeric_var)) {
    stop("numeric_var must be one of: ", paste(allowed_numeric_vars, collapse = ","))
  }
  col_names_tibble <- colnames(x)
  
  if(!(numeric_var %in% col_names_tibble)) {
    stop("numeric_var is not part of available tibble columns: ", paste(col_names_tibble, collapse = ","))
  }
  
  categorical_var <- match_allowed_value(categorical_var, allowed_categorical_vars)
  if(is.na(categorical_var)) {
    stop("categorical_var must be one of: ", paste(allowed_categorical_vars, collapse = ","))
  }
  
   if(!(categorical_var %in% col_names_tibble)) {
    stop("categorical_var is not part of available tibble columns: ", paste(col_names_tibble, collapse = ","))
   }
  
  ggplot(x, aes(x = get(categorical_var), y = get(numeric_var), weight = PWGTP)) + 
    geom_boxplot() + 
    labs(x = paste0(categorical_var, "(", get_label_for_var(categorical_var), ")"),
         y = paste0(numeric_var, "(", get_label_for_var(numeric_var), ")"),
         title = paste(numeric_var, " by ", categorical_var),
         subtitle = "Weighted by PWGTP (survey weight)",
         caption = "Source: Census PUMS data")
  
}

Investigation-2 GRPIP vs SEX

One immediate things that stands out in GRPIP vs SEX plot is the lower fence of IQR actually starts at 0

This means there are people who do not pay rent at all and their share of NC population is more than 25 %. Per boxplot definition since at least 25% of the population pays no rent at all, the 25th percentile (Q1) of GRPIP is exactly 0. Because GRPIP can’t go below 0, there is no room for a lower whisker to extend. It collapses to the same point as the box edge.
